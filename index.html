If they mix languages, respond naturally in the same mix." : `Reply in ${String(language).slice(0, 40)}.`; const response = await client.responses.create({ model: process.env.OPENAI_MODEL || "gpt-5.5", instructions: "You are UNIQUE AI, a friendly, helpful, age-appropriate multilingual learning assistant. " + languageInstruction + " Give clear, accurate answers. For educational questions, explain step by step at an appropriate level. " + "Do not claim to have capabilities you do not have.", input: [ ...safeHistory, { role: "user", content: message.trim() } ] }); res.json({ reply: response.output_text || "Sorry, I could not form a reply. Please try again." }); } catch (error) { console.error("Chat API error:", error?.message || error); res.status(500).json({ error: "AI service is temporarily unavailable. Please try again later." }); } }); app.get("/health", (_req, res) => res.json({ ok: true, app: "UNIQUE AI" })); app.listen(port, () => { console.log(`UNIQUE AI server running on port ${port}`); });
